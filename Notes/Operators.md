```python

Arithmetic
+     # addition
-     # subtraction
*     # multiplication
/     # division (gives decimal)
//    # floor division (removes decimal)
%     # remainder
**    # power

Comparison
==    # equal to
!=    # not equal to
>     # greater than
<     # less than
>=    # greater than or equal to
<=    # less than or equal to

Logical
and   # both must be True
or    # at least one must be True
not   # reverses True/False

Assignment
=     # assign a value
+=    # add and assign
-=    # subtract and assign
*=    # multiply and assign
//=   # floor divide and assign
%=    # remainder and assign

Slicing
x[start : stop : step]

x[::1]      # whole thing, forwards
x[::-1]     # whole thing, backwards
x[1:4]      # index 1 up to (not including) 4
x[:4]       # beginning → index 4
x[2:]       # index 2 → end
x[::2]      # every 2nd element
x[::-1]     # reverse
x[1:5:2]    # index 1 → 5, every 2nd element
```
